import bpy
import random
import math

def create_crystalline_material():
    mat = bpy.data.materials.new(name="Crystalline_Material")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = (0.8, 0.9, 1.0, 1.0)
        bsdf.inputs['Metallic'].default_value = 0.1
        bsdf.inputs['Roughness'].default_value = 0.1
        bsdf.inputs['IOR'].default_value = 1.5
        bsdf.inputs['Transmission Weight'].default_value = 0.9
    return mat

def create_soil_material():
    mat = bpy.data.materials.new(name="Soil_Material")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = (0.1, 0.05, 0.03, 1.0)
        bsdf.inputs['Roughness'].default_value = 0.9
    return mat

def create_plant_material(color):
    mat = bpy.data.materials.new(name="Plant_Material")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        bsdf.inputs['Subsurface Weight'].default_value = 0.1
        bsdf.inputs['Subsurface Radius'].default_value = (0.2, 0.2, 0.1)
        bsdf.inputs['Roughness'].default_value = 0.8
    return mat

def create_crystalline_structures(material):
    locations = [(0, 0, 5), (8, -5, 8), (-7, 6, 12)]
    for i, loc in enumerate(locations):
        if i % 2 == 0:
            bpy.ops.mesh.primitive_cube_add(size=10, location=loc)
        else:
            bpy.ops.mesh.primitive_ico_sphere_add(radius=6, subdivisions=3, location=loc)
        
        obj = bpy.context.active_object
        obj.name = f"Crystalline_Structure_{i}"
        
        mod_gn = obj.modifiers.new(name="Displace", type='NODES')
        node_group = bpy.data.node_groups.new(name=f"GeoNodes_Displace_{i}", type='GeometryNodeTree')
        mod_gn.node_group = node_group
        
        nodes = node_group.nodes
        links = node_group.links
        
        for node in nodes:
            nodes.remove(node)

        group_input = nodes.new(type='NodeGroupInput')
        group_output = nodes.new(type='NodeGroupOutput')
        set_position = nodes.new(type='GeometryNodeSetPosition')
        voronoi = nodes.new(type='GeometryNodeTexVoronoi')
        normal = nodes.new(type='GeometryNodeInputNormal')
        scale_vec = nodes.new(type='ShaderNodeVectorMath')
        scale_vec.operation = 'SCALE'
        subtract = nodes.new('ShaderNodeMath')
        subtract.operation = 'SUBTRACT'
        subtract.inputs[1].default_value = 0.5
        multiply_strength = nodes.new('ShaderNodeMath')
        multiply_strength.operation = 'MULTIPLY'

        group_input.location = (-600, 0)
        group_output.location = (600, 0)
        set_position.location = (400, 0)
        voronoi.location = (-400, -200)
        normal.location = (-200, 100)
        scale_vec.location = (200, 0)
        subtract.location = (-200, -200)
        multiply_strength.location = (0, -100)

        if not node_group.interface.items_tree.get("Geometry"):
            node_group.interface.new_socket(name="Geometry", in_out="INPUT", socket_type='NodeSocketGeometry')
        if not node_group.interface.items_tree.get("Geometry"):
            node_group.interface.new_socket(name="Geometry", in_out="OUTPUT", socket_type='NodeSocketGeometry')
            
        strength_socket = node_group.interface.new_socket(name="Strength", in_out="INPUT", socket_type='NodeSocketFloat')
        strength_socket.default_value = random.uniform(1.0, 2.5)
        strength_socket.min_value = 0.0
        strength_socket.max_value = 10.0
        strength_socket.attribute_domain = 'POINT'

        voronoi.inputs['Scale'].default_value = random.uniform(0.5, 1.5)
        
        links.new(group_input.outputs['Geometry'], set_position.inputs['Geometry'])
        links.new(set_position.outputs['Geometry'], group_output.inputs['Geometry'])
        links.new(voronoi.outputs['Distance'], subtract.inputs[0])
        links.new(subtract.outputs['Value'], multiply_strength.inputs[0])
        links.new(group_input.outputs['Strength'], multiply_strength.inputs[1])
        links.new(multiply_strength.outputs['Value'], scale_vec.inputs['Scale'])
        links.new(normal.outputs['Normal'], scale_vec.inputs['Vector'])
        links.new(scale_vec.outputs['Vector'], set_position.inputs['Offset'])

        obj.data.materials.append(material)

def create_plant(name, location, plant_type):
    obj = None
    if plant_type == 'tree':
        bpy.ops.mesh.primitive_cylinder_add(vertices=8, radius=0.2, depth=3, location=location)
        trunk = bpy.context.active_object
        trunk.name = f"{name}_Trunk"
        trunk.data.materials.append(create_plant_material((0.3, 0.15, 0.05, 1.0)))
        
        bpy.ops.mesh.primitive_ico_sphere_add(radius=1.5, subdivisions=1, location=(location[0], location[1], location[2] + 2.5))
        leaves = bpy.context.active_object
        leaves.name = f"{name}_Leaves"
        leaves.data.materials.append(create_plant_material((0.1, 0.4, 0.1, 1.0)))
        
        leaves.parent = trunk
        bpy.context.view_layer.objects.active = trunk
        bpy.ops.object.select_all(action='DESELECT')
        trunk.select_set(True)
        leaves.select_set(True)
        bpy.ops.object.join()
        obj = bpy.context.active_object

    elif plant_type == 'shrub':
        bpy.ops.mesh.primitive_ico_sphere_add(radius=random.uniform(0.5, 1.2), subdivisions=1, location=location)
        obj = bpy.context.active_object
        obj.data.materials.append(create_plant_material((0.2, 0.5, 0.15, 1.0)))
        
    elif plant_type == 'flower':
        bpy.ops.mesh.primitive_cylinder_add(vertices=6, radius=0.05, depth=0.8, location=location)
        stem = bpy.context.active_object
        stem.name = f"{name}_Stem"
        stem.data.materials.append(create_plant_material((0.1, 0.6, 0.1, 1.0)))
        
        bpy.ops.mesh.primitive_cone_add(vertices=12, radius1=0.3, depth=0.1, location=(location[0], location[1], location[2] + 0.4))
        petal = bpy.context.active_object
        petal.name = f"{name}_Petal"
        petal.data.materials.append(create_plant_material( (random.uniform(0.5,1.0), random.uniform(0.1,0.5), random.uniform(0.5,1.0), 1.0) ))
        
        petal.parent = stem
        bpy.context.view_layer.objects.active = stem
        bpy.ops.object.select_all(action='DESELECT')
        stem.select_set(True)
        petal.select_set(True)
        bpy.ops.object.join()
        obj = bpy.context.active_object
        
    elif plant_type == 'vine':
        bpy.ops.curve.primitive_bezier_curve_add(location=location)
        curve_obj = bpy.context.active_object
        curve_obj.data.dimensions = '3D'
        curve_obj.data.bevel_depth = 0.02
        curve_obj.data.bevel_resolution = 1
        spline = curve_obj.data.splines[0]
        spline.bezier_points[0].co = (0, 0, 0)
        spline.bezier_points[1].co = (0.5, 0.5, 2)
        spline.type = 'POLY'
        bpy.ops.object.convert(target='MESH')
        obj = bpy.context.active_object
        obj.data.materials.append(create_plant_material((0.1, 0.3, 0.1, 1.0)))

    if obj:
        obj.name = name
        wind_mod = obj.modifiers.new(name="Wind", type='SIMPLE_DEFORM')
        wind_mod.deform_method = 'BEND'
        wind_mod.deform_axis = 'X'
        driver = wind_mod.driver_add('angle')
        driver.driver.expression = f"sin(frame * 0.1 + {random.uniform(0, 2*math.pi)}) * {random.uniform(0.05, 0.15)}"
    
    return obj

plant_collection_name = "Plant_Models"
if plant_collection_name not in bpy.data.collections:
    plant_collection = bpy.data.collections.new(plant_collection_name)
    bpy.context.scene.collection.children.link(plant_collection)
else:
    plant_collection = bpy.data.collections[plant_collection_name]
    
if plant_collection.name in bpy.context.view_layer.layer_collection.children:
    bpy.context.view_layer.layer_collection.children[plant_collection.name].exclude = True

plant_types = ['tree', 'shrub', 'flower', 'vine']
num_plant_species = 15
plant_objects = []
for i in range(num_plant_species):
    plant_type = random.choice(plant_types)
    loc = (i * 2.5 - (num_plant_species * 1.25), -20, 0)
    plant = create_plant(f"Plant_{i}", loc, plant_type)
    if plant:
        plant_objects.append(plant)
        for ob in bpy.context.selected_objects:
             for coll in ob.users_collection:
                coll.objects.unlink(ob)
             plant_collection.objects.link(ob)

crystal_mat = create_crystalline_material()
soil_mat = create_soil_material()

create_crystalline_structures(crystal_mat)

garden_locations = [(0, 0, 10), (8, -5, 11), (-7, 6, 15.5)]
garden_scales = [(5, 5, 1), (3, 3, 1), (4, 2, 1)]

for i, (loc, scale) in enumerate(zip(garden_locations, garden_scales)):
    bpy.ops.mesh.primitive_plane_add(size=1, location=loc)
    garden_plane = bpy.context.active_object
    garden_plane.name = f"Garden_Area_{i}"
    garden_plane.scale = scale
    garden_plane.data.materials.append(soil_mat)

    ps_name = f"Foliage_{i}"
    if ps_name in garden_plane.particle_systems:
        ps = garden_plane.particle_systems[ps_name]
    else:
        ps = garden_plane.particle_systems.new(name=ps_name)
        
    settings = ps.settings
    if settings is None:
        settings = bpy.data.particle_settings.new(name=ps_name)
        ps.settings = settings

    settings.type = 'HAIR'
    settings.use_advanced_hair = True
    settings.count = 20 * int(scale[0]*scale[1]/5)
    settings.hair_length = 1.0

    settings.render_type = 'COLLECTION'
    settings.instance_collection = plant_collection
    settings.use_collection_pick_random = True
    settings.particle_size = random.uniform(0.3, 0.8)
    settings.size_random = random.uniform(0.5, 1.0)
    
    settings.use_rotations = True
    settings.rotation_mode = 'GLOB_Z'
    settings.rotation_factor_random = 1.0
    settings.phase_factor_random = 2.0
